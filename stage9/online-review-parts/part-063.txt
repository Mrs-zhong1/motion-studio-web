const imageBlobCache=new Map();
async function cachedImageBlob(url){
 if(!imageBlobCache.has(url)){
   imageBlobCache.set(url,(async()=>{
     const response=await fetch(url,{mode:'cors',cache:'force-cache'});
     if(!response.ok)throw Error('image HTTP '+response.status);
     const blob=await response.blob();
     if(!blob.type.startsWith('image/'))throw Error('invalid image');
     return blob;
   })());
 }
 return imageBlobCache.get(url);
}
async function photoToInline(url,variant=0){
 try{
   const blob=await cachedImageBlob(url);
   if(!blob.type.startsWith('image/'))throw Error('invalid image');
   const image=await createImageBitmap(blob);
   const factor=Math.min(1,840/Math.max(image.width,image.height));
   const canvas=document.createElement('canvas');
   canvas.width=Math.max(1,Math.round(image.width*factor));
   canvas.height=Math.max(1,Math.round(image.height*factor));
   const ctx=canvas.getContext('2d');
   if(variant===0)ctx.drawImage(image,0,0,canvas.width,canvas.height);
   else{
     const crops=[
       {scale:.83,x:.12,y:.15},
       {scale:.72,x:.82,y:.72},
       {scale:.9,x:.62,y:.22},
       {scale:.76,x:.22,y:.78},
       {scale:.88,x:.5,y:.5}
     ];
     const crop=crops[(variant-1)%crops.length];
     const w=image.width*crop.scale,h=image.height*crop.scale;
     const x=(image.width-w)*crop.x,y=(image.height-h)*crop.y;
     ctx.drawImage(image,x,y,w,h,0,0,canvas.width,canvas.height);
   }
   image.close?.();
   const data=canvas.toDataURL('image/jpeg',.64);
   return data.length<1500000?data:'';
 }catch(err){console.warn('Demo asset unavailable:',url,err);return ''}
}

async function importOriginalAIFolder(files){
  const selections=Array.from(files||[]);
  const manifestFile=selections.find(f=>f.name==='manifest.json');
  if(!manifestFile)throw Error('请选择包含 manifest.json 和 39 张图片的 ai-assets 文件夹');
  if(!globalThis.crypto?.subtle)throw Error('此页面需要 HTTPS 或 localhost 以进行 SHA-256 校验');
  const manifest=JSON.parse(await manifestFile.text());
  if(!Array.isArray(manifest.assets)||manifest.assets.length!==39)throw Error('AI 素材必须恰好包含 39 个章节');
  const expected=Object.fromEntries(templateCatalog.map(t=>[t.id,t.sections.length]));
  const fileByName=new Map(selections.filter(f=>f!==manifestFile).map(f=>[f.name,f]));
  const next={};const seen=new Set();
  const sha256=async bytes=>Array.from(new Uint8Array(await crypto.subtle.digest('SHA-256',bytes))).map(b=>b.toString(16).padStart(2,'0')).join('');
  for(const item of manifest.assets){
    if(!Object.hasOwn(expected,item.template)||!Number.isInteger(item.chapter)||
       item.chapter<1||item.chapter>expected[item.template]||
       !/^[a-z0-9-]+\.(jpg|jpeg|png|webp)$/i.test(item.file)||
       item.source!=='AI_GENERATED'||item.commercialUseTermsChecked!==true)
      throw Error('AI 文件来源、使用条款或章节编号不完整');
    const key=item.template+':'+item.chapter;
    if(seen.has(key))throw Error('素材章节重复：'+key);
    seen.add(key);
    const file=fileByName.get(item.file);
    if(!file||file.size<1000||file.size>5_000_000)throw Error('素材文件不存在或尺寸异常：'+item.file);
    const bytes=await file.arrayBuffer();
    const digest=await sha256(bytes);
    if(digest!==item.sha256)throw Error('素材 SHA-256 不匹配：'+item.file);
    const ext=item.file.toLowerCase().split('.').pop();
    const mime=ext==='png'?'image/png':ext==='webp'?'image/webp':'image/jpeg';
    const b64=await new Promise((resolve,reject)=>{
      const reader=new FileReader();
      reader.onload=()=>resolve(String(reader.result).split(',')[1]||'');
      reader.onerror=()=>reject(Error('无法读取 '+item.file));
      reader.readAsDataURL(new Blob([bytes],{type:mime}));
    });
    (next[item.template] ||= [])[item.chapter-1]={
       media:'data:'+mime+';base64,'+b64,
       source:'ai-generated://'+item.template+'/'+item.chapter+'/'+digest.slice(0,16)
    };
  }
  for(const [id,count] of Object.entries(expected))
    if(next[id]?.length!==count||next[id].some(x=>!x))throw Error('模板章节不完整：'+id);
  // Commit as one atomic state update only after every hash and chapter passes.
  for(const [id,entries] of Object.entries(next)){
    templateOfflinePhotoBundles[id]=entries;
    templateThumbnailImages[id]=entries[0].media;
  }
  templatePhotoCache.clear();
  templatePhotoPending.clear();
  templateAssetAuditCache.clear();
  renderTemplateGrid();
  return {templateCount:Object.keys(next).length,chapterCount:seen.size};
}

function builtInHeadphoneGlb(){

 // R9-03: original, redistributable parametric headphone demonstration.
 // Smooth closed curved surfaces replace the former five scaled cubes.
 // Four genuine PBR primitives share a single valid GLB binary buffer.
 const materials=[
  {name:'Ceramic shell',pbrMetallicRoughness:{baseColorFactor:[.20,.31,.35,1],metallicFactor:.46,roughnessFactor:.28}},
  {name:'Soft-touch cushion',pbrMetallicRoughness:{baseColorFactor:[.060,.085,.10,1],metallicFactor:.02,roughnessFactor:.81}},

 ];

 const push=(mat,point,normal)=>{const a=sets[mat];a.p.push(...point);a.n.push(...normal);return a.p.length/3-1};
 const triangle=(mat,a,b,c)=>sets[mat].i.push(a,b,c);
 const grid=(mat,ringA,ringB,closed)=>{
   const N=ringA.length;
   for(let j=0;j<(closed?N:N-1);j++){
     const k=(j+1)%N;
     triangle(mat,ringA[j],ringA[k],ringB[k]);
     triangle(mat,ringA[j],ringB[k],ringB[j]);
   }
 };
 const normalize=(x,y,z)=>{const d=Math.hypot(x,y,z)||1;return [x/d,y/d,z/d]};
 // Smooth elliptic ear cups and soft rim rings: lateral axis = X.
 function cup(side,mat,cx,cy,cz,rx,ry,rz,segments=64,rings=24){
   let prev=null;const caps=[];
   for(let j=0;j<=rings;j++){
     const theta=.0005+(Math.PI-.001)*j/rings;
     const ring=[];
     for(let k=0;k<segments;k++){
       const phi=2*Math.PI*k/segments;
       const x=side*(cx+rx*Math.cos(theta)),
             y=cy+ry*Math.sin(theta)*Math.cos(phi),
             z=cz+rz*Math.sin(theta)*Math.sin(phi);
       const normal=normalize(side*Math.cos(theta)/rx,Math.sin(theta)*Math.cos(phi)/ry,Math.sin(theta)*Math.sin(phi)/rz);
       ring.push(push(mat,[x,y,z],normal));
     }
     if(prev)grid(mat,side===1?prev:ring,side===1?ring:prev,true);
     prev=ring;
   }
 }
 // A swept tubular curve, used for headband and yokes with smoothed normals.
 function tube(mat,path,radius,segments=12){
   let prior=null;
   for(let i=0;i<path.length;i++){
     const p=path[i],before=path[Math.max(0,i-1)],after=path[Math.min(path.length-1,i+1)];
     const tangent=normalize(after[0]-before[0],after[1]-before[1],after[2]-before[2]);
     let up=[0,0,1];
     if(Math.abs(tangent[2])>.89)up=[0,1,0];
     const u=normalize(up[1]*tangent[2]-up[2]*tangent[1],up[2]*tangent[0]-up[0]*tangent[2],up[0]*tangent[1]-up[1]*tangent[0]);
     const v=normalize(tangent[1]*u[2]-tangent[2]*u[1],tangent[2]*u[0]-tangent[0]*u[2],tangent[0]*u[1]-tangent[1]*u[0]);
     const ring=[];
     for(let j=0;j<segments;j++){
       const angle=j*Math.PI*2/segments;const nx=u[0]*Math.cos(angle)+v[0]*Math.sin(angle),ny=u[1]*Math.cos(angle)+v[1]*Math.sin(angle),nz=u[2]*Math.cos(angle)+v[2]*Math.sin(angle);
       ring.push(push(mat,[p[0]+nx*radius,p[1]+ny*radius,p[2]+nz*radius],[nx,ny,nz]));
     }
     if(prior)grid(mat,prior,ring,true);
     prior=ring;
   }
 }
 // Closed over-ear housing, inset cushions and centre speaker mesh.
 for(const side of [-1,1]){
   cup(side,0,.78,-.35,0,.25,.55,.37);
   cup(side,1,.57,-.35,0,.125,.48,.32);
   cup(side,3,.46,-.35,0,.029,.37,.245,64,12);
   const hinge=[];
   for(let j=0;j<=12;j++){
     const t=j/12;hinge.push([side*(.80-.08*t),.12+.43*t,0]);
   }
   tube(2,hinge,.065);
   // A subtle metallic cup gasket/outer ring.
   const rim=[];
   for(let j=0;j<=72;j++){
     const t=2*Math.PI*j/72;
     rim.push([side*1.018,-.35+Math.cos(t)*.46,Math.sin(t)*.32]);
   }
   tube(2,rim,.018,8);
 }
 // True curved headband with a wider, compliant inner padding.
 const headband=[],inside=[];
 for(let k=0;k<=88;k++){
   const a=Math.PI-.08-(Math.PI-.16)*k/88;
   headband.push([.79*Math.cos(a),.55+1.05*Math.sin(a),0]);
   if(k>=9&&k<=79)inside.push([.79*Math.cos(a),.49+1.01*Math.sin(a),0]);
 }
 tube(2,headband,.065,14);
 tube(1,inside,.086,14);
 const bin=[],views=[],accessors=[],primitives=[];
 const align4=()=>{while(bin.length%4)bin.push(0)};
 const viewFor=(bytes,target)=>{
   align4();const offset=bin.length;for(const b of bytes)bin.push(b);
   const index=views.length;views.push({buffer:0,byteOffset:offset,byteLength:bytes.length,target});return index;
 };
 const toBytes=(arr,kind)=>{
   const bytes=new Uint8Array(arr.length*(kind==='float'?4:2));
   const view=new DataView(bytes.buffer);
   arr.forEach((v,i)=>kind==='float'?view.setFloat32(i*4,v,true):view.setUint16(i*2,v,true));
   return bytes;
 };
 const accessor=(bufferView,componentType,count,type,limits)=>{
   const x={bufferView,componentType,count,type,...(limits||{})};accessors.push(x);return accessors.length-1;
 };
 for(let mat=0;mat<sets.length;mat++){
   const a=sets[mat];if(!a.i.length)continue;
   if(a.p.length/3>65535)throw Error('R9 mesh exceeds Uint16 index range');
   const posView=viewFor(toBytes(a.p,'float'),34962);
   const normView=viewFor(toBytes(a.n,'float'),34962);
   const idxView=viewFor(toBytes(a.i,'ushort'),34963);
   let lo=[Infinity,Infinity,Infinity],hi=[-Infinity,-Infinity,-Infinity];
   for(let i=0;i<a.p.length;i+=3)for(let j=0;j<3;j++){lo[j]=Math.min(lo[j],a.p[i+j]);hi[j]=Math.max(hi[j],a.p[i+j])}
   const pos=accessor(posView,5126,a.p.length/3,'VEC3',{min:lo,max:hi});
   const norm=accessor(normView,5126,a.n.length/3,'VEC3');
   const indices=accessor(idxView,5123,a.i.length,'SCALAR');
   primitives.push({attributes:{POSITION:pos,NORMAL:norm},indices,material:mat,mode:4});
 }

 const gltf={asset:{version:'2.0',generator:'Motion Studio original procedural R9 curved headphone'},
  scene:0,scenes:[{nodes:[0]}],nodes:[{mesh:0,name:'Headphone / original parametric study'}],

}
