/* R15-02: user-owned batch chapter photography, compressed for standalone exports.
   Keep the existing template motion/3D sections intact. Atomic update on success. */
async function r15CompressPhoto(file){
 if(!['image/png','image/jpeg','image/webp'].includes(file.type)||file.size>15000000||file.size<=0)throw Error('图片类型不支持或超过15MB：'+file.name);
 const bitmap=await createImageBitmap(file);
 try{
  const scale=Math.min(1,1400/Math.max(bitmap.width,bitmap.height));
  const canvas=document.createElement('canvas');
  canvas.width=Math.max(1,Math.round(bitmap.width*scale));
  canvas.height=Math.max(1,Math.round(bitmap.height*scale));
  const ctx=canvas.getContext('2d');
  if(!ctx)throw Error('浏览器图像处理不可用');
  ctx.fillStyle='#f9f9f7';ctx.fillRect(0,0,canvas.width,canvas.height);
  ctx.drawImage(bitmap,0,0,canvas.width,canvas.height);
  for(const quality of [.84,.72,.6,.48]){
   const jpeg=canvas.toDataURL('image/jpeg',quality);
   if(jpeg.length<2300000)return jpeg;
  }
  throw Error('图片压缩后仍超过导出限制：'+file.name);
 }finally{bitmap.close?.()}
}
